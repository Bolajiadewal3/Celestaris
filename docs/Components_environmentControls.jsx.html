/**
 * @category Utility
 * @description Provides controls for managing the environment settings.
 */
import { useEnvironmentStore } from "../Store/useEnvironmentStore";

/**
 * Renders a UI panel for controlling the environment settings, including time of day and weather.
 * @category Environment
 */
export function EnvironmentControls() {
  const {
    manualTime,
    setManualTime,
    isOverriding,
    setOverriding,
    weather,
    setWeather,
    setWeatherMode,
  } = useEnvironmentStore();

  return (
    <div
      style={{
        position: "absolute",
        top: 100,
        right: 20,
        zIndex: 100,
        width: "220px",
        background: "rgba(10, 12, 22, 0.9)",
        backdropFilter: "blur(12px)",
        WebkitBackdropFilter: "blur(12px)",
        padding: "16px",
        borderRadius: "14px",
        border: "1px solid rgba(255, 255, 255, 0.12)",
        color: "#fff",
        fontFamily: "Orbitron, sans-serif",
        boxShadow: "0 12px 32px rgba(0, 0, 0, 0.6)",
        display: "flex",
        flexDirection: "column",
        gap: "16px",
      }}
    >
      {/* SECTION 1: TIME OF DAY */}
      <div>
        <div
          style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            marginBottom: "8px",
          }}
        >
          <span style={{ fontSize: "12px", fontWeight: 600, color: "#cbd5e1" }}>
            TIME OF DAY
          </span>
          <span style={{ fontSize: "12px", color: "#a855f7", fontWeight: 700 }}>
            {manualTime.toFixed(1)}:00
          </span>
        </div>

        <input
          type="range"
          min="0"
          max="24"
          step="0.5"
          value={manualTime}
          onMouseDown={() => setOverriding(true)}
          onChange={(e) => setManualTime(Number.parseFloat(e.target.value))}
          style={{
            width: "100%",
            marginBottom: "8px",
            cursor: "pointer",
            accentColor: "#a855f7",
          }}
        />

        {/* Time Mode Segmented Switch */}
        <div
          style={{
            display: "flex",
            background: "rgba(255, 255, 255, 0.05)",
            padding: "2px",
            borderRadius: "8px",
            border: "1px solid rgba(255, 255, 255, 0.08)",
          }}
        >
          <button
            onClick={() => setOverriding(false)}
            style={{
              flex: 1,
              padding: "5px 0",
              background: !isOverriding ? "#a855f7" : "transparent",
              border: "none",
              borderRadius: "6px",
              color: !isOverriding ? "#fff" : "#94a3b8",
              cursor: "pointer",
              fontSize: "11px",
              fontWeight: !isOverriding ? 600 : 400,
              transition: "all 0.2s ease",
            }}
          >
            Auto Cycle
          </button>
          <button
            onClick={() => setOverriding(true)}
            style={{
              flex: 1,
              padding: "5px 0",
              background: isOverriding ? "#a855f7" : "transparent",
              border: "none",
              borderRadius: "6px",
              color: isOverriding ? "#fff" : "#94a3b8",
              cursor: "pointer",
              fontSize: "11px",
              fontWeight: isOverriding ? 600 : 400,
              transition: "all 0.2s ease",
            }}
          >
            Manual
          </button>
        </div>
      </div>

      <div style={{ height: "1px", background: "rgba(255, 255, 255, 0.1)" }} />

      {/* SECTION 2: WEATHER CONTROL */}
      <div>
        <div
          style={{
            fontSize: "12px",
            fontWeight: 600,
            color: "#cbd5e1",
            marginBottom: "8px",
          }}
        >
          WEATHER OVERRIDE
        </div>

        {/* Weather Segmented Controls */}
        <div
          style={{
            display: "flex",
            gap: "6px",
            marginBottom: "8px",
          }}
        >
          <button
            onClick={() => {
              setWeatherMode("manual");
              setWeather("clear");
            }}
            style={{
              flex: 1,
              padding: "6px 0",
              background:
                weather === "clear" ? "#a855f7" : "rgba(255,255,255,0.05)",
              border:
                "1px solid " +
                (weather === "clear" ? "#a855f7" : "rgba(255,255,255,0.08)"),
              borderRadius: "6px",
              color: weather === "clear" ? "#fff" : "#94a3b8",
              cursor: "pointer",
              fontSize: "11px",
              fontWeight: weather === "clear" ? 600 : 400,
              transition: "all 0.2s ease",
            }}
          >
            ☀️ Clear
          </button>

          <button
            onClick={() => {
              setWeatherMode("manual");
              setWeather("rain");
            }}
            style={{
              flex: 1,
              padding: "6px 0",
              background:
                weather === "rain" ? "#a855f7" : "rgba(255,255,255,0.05)",
              border:
                "1px solid " +
                (weather === "rain" ? "#a855f7" : "rgba(255,255,255,0.08)"),
              borderRadius: "6px",
              color: weather === "rain" ? "#fff" : "#94a3b8",
              cursor: "pointer",
              fontSize: "11px",
              fontWeight: weather === "rain" ? 600 : 400,
              transition: "all 0.2s ease",
            }}
          >
            🌧️ Rain
          </button>
        </div>

        {/* Dynamic Reset Button */}
        <button
          onClick={() => {
            setWeatherMode("auto");
            setWeather("clear");
          }}
          style={{
            width: "100%",
            padding: "6px 0",
            background: "rgba(168, 85, 247, 0.12)",
            border: "1px solid rgba(168, 85, 247, 0.3)",
            borderRadius: "6px",
            color: "#d8b4fe",
            cursor: "pointer",
            fontSize: "11px",
            fontWeight: 600,
            transition: "all 0.2s ease",
          }}
        >
          🔄 Dynamic Sync
        </button>
      </div>
    </div>
  );
}
